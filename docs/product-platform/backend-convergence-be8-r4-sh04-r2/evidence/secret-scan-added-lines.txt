+# BE8 Architecture — Universal Security Discovery / Planner
+
+## Scope
+
+BE8 adds a non-destructive, read-only security discovery and planning layer. It determines what a governed target appears to be, which security capability categories apply, and which future assessment phases need runtime, build, deployment, network, or credential authority.
+
+BE8 does not execute arbitrary scanners, mutate CI pipelines, install dependencies, run package-manager scripts, perform network requests, accept browser-declared stack truth, or expose shell/Docker execution surfaces.
+
+## Components
+
+- `frontend/src/server/product/security/types.ts` defines stack detection, capability, applicability, and assessment-plan contracts.
+- `frontend/src/server/product/security/discovery.ts` performs static metadata detection with provenance.
+- `frontend/src/server/product/security/capabilities.ts` defines the server-owned capability registry.
+- `frontend/src/server/product/security/planner.ts` maps detected facts into deterministic assessment plans.
+- `frontend/src/server/product/security/service.ts` enforces product workspace scope.
+- `frontend/src/app/api/product/security/**` exposes bounded discovery and plan endpoints.
+
+## Discovery model
+
+The browser supplies only bounded discovery selectors: `projectId`, optional `targetId`, and optional `assessmentIntent`. It cannot supply source files, file metadata, source revision, detected stack, provenance, or a prior discovery object.
+
+A server-owned `SecuritySourceAdapter` resolves the authenticated Principal/workspace/project/target to the trusted source snapshot. The default production adapter fails closed until the governed registered-project source integration is available. Paths from that trusted snapshot must be workspace-relative, normalized, non-absolute, non-empty, and non-escaping. Symlinks, binary files, oversized manifests, malformed manifests, and malicious paths are reported as unsupported facts rather than silently trusted.
+
+Every detected technology carries source type, path, confidence, and detector version. Caller-declared stack values are rejected rather than treated as truth.
+
+## Planner model
+
+The planner is deterministic. It creates no execution authority and emits capability states such as `APPLICABLE`, `REQUIRES_RUNTIME`, `REQUIRES_BUILD`, `REQUIRES_DEPLOYMENT`, `REQUIRES_CREDENTIAL`, `REQUIRES_NETWORK`, `NOT_APPLICABLE`, and `UNKNOWN`. Unknown facts remain unknown.
+
+## Future handoff
+
+BE9-BE12 may consume BE8 plans to perform governed scanner execution or pipeline synthesis. BE8 only creates the planning contract.

+# BE8 Security Analysis
+
+## Authority boundaries
+
+BE8 keeps planner authority separate from execution authority. It does not introduce `run_security_tool`, `run_scanner`, `docker_run`, `shell_exec`, or any generic command execution API.
+
+Browser input may request discovery using only a bounded `projectId`, optional `targetId`, and optional `assessmentIntent`. The browser cannot supply source files, file metadata, source revision, detected stack, provenance, or a discovery result.
+
+Source revision and file bytes/metadata come only from the server-owned `SecuritySourceAdapter`. That adapter is responsible for resolving the authenticated user's governed registered project/target source. The default production adapter intentionally fails closed as unavailable until that governed integration exists.
+
+The server decides detected technologies and capability applicability.
+
+## Read-only constraints
+
+Discovery is static and read-only. BE8 does not run package scripts, perform dependency installation, build the target merely for discovery, execute scanners, or perform network requests.
+
+## Path and parser safety
+
+Workspace paths are normalized and rejected if absolute, escaping, empty, malformed, too long, or containing NUL/backslash escape shapes. Symlinks are unsupported. Oversized manifests are refused. Malformed package manifests become unsupported facts rather than fallback assumptions.
+
+## Multi-tenant isolation
+
+The service requires a product session with a workspace. Project IDs that attempt path-like or escaping forms are rejected. BE8 does not expose cross-workspace search or browser-supplied organization/workspace authority.
+
+## Live gaps
+
+BE8 does not qualify live Postgres, Keycloak/Entra, OpenBao, STT, audio capture, transcript store, realtime storage, or monitoring/runtime dependencies. Those remain open live qualification gates.

+function normalizeWorkspacePath(path: string): string | undefined {
+  if (path.length === 0 || path.length > MAX_PATH_LENGTH) return undefined;
+  if (path.startsWith("/") || path.includes("\\") || path.includes("\0")) return undefined;
+  const normalized = posix.normalize(path);
+  if (normalized === "." || normalized.startsWith("../") || normalized === "..") return undefined;
+  if (normalized.split("/").some((part) => part === "" || part === "." || part === "..")) return undefined;
+  return normalized;

+export function discoverSecurityStack(snapshot: TrustedSecuritySourceSnapshot): SecurityDiscoveryResult {
+  assertBoundedIdentity(snapshot);
+  if (snapshot.files.length > MAX_SOURCE_FILES) throw new Error("security_discovery_file_count_exceeded");
+
+  const map = new Map<string, DetectedTechnology>();
+  const unsupported: DiscoveryUnsupportedFact[] = [];
+  let totalBytes = 0;
+
+  for (const file of snapshot.files) {
+    const content = file.content ?? "";
+    const actualBytes = Buffer.byteLength(content, "utf8");
+    const metadataBytes = Number.isSafeInteger(file.sizeBytes) && (file.sizeBytes ?? 0) >= 0 ? file.sizeBytes ?? 0 : 0;
+    const boundedBytes = Math.max(actualBytes, metadataBytes);
+    totalBytes += boundedBytes;
+    if (totalBytes > MAX_TOTAL_DISCOVERY_BYTES) throw new Error("security_discovery_total_bytes_exceeded");
+
+    const path = normalizeWorkspacePath(file.path);
+    if (!path) {
+      unsupported.push({ path: file.path.slice(0, MAX_PATH_LENGTH + 1), reason: "MALICIOUS_PATH" });
+      continue;
+    }
+    if (file.kind === "symlink") {
+      unsupported.push({ path, reason: "SYMLINK_UNSUPPORTED" });
+      continue;
+    }
+    if (file.binary) {
+      unsupported.push({ path, reason: "BINARY_UNPARSEABLE" });
+      continue;
+    }
+
+    const name = basename(path).toLowerCase();
+    const isManifest = ["package.json", "package-lock.json", "pnpm-lock.yaml", "yarn.lock", "requirements.txt", "pyproject.toml", "go.mod"].includes(name) || path.endsWith(".csproj") || path.endsWith(".sln");
+    if (isManifest && boundedBytes > MAX_MANIFEST_BYTES) {
+      unsupported.push({ path, reason: "OVERSIZED_MANIFEST" });
+      continue;
+    }
+    if (boundedBytes > MAX_SINGLE_FILE_BYTES) {
+      unsupported.push({ path, reason: "OVERSIZED_SOURCE_FILE" });
+      continue;
+    }
+
+    if (name === "package.json") inspectPackageJson(path, content, map, unsupported);
+    if (name === "package-lock.json" || name === "pnpm-lock.yaml" || name === "yarn.lock") addTechnology(map, tech("node-lockfile", "Node lockfile", "deployment", [provenance("lockfile", path, name)]));
+    if (name === "requirements.txt" || name === "pyproject.toml") addTechnology(map, tech("python", "Python", "language", [provenance("dependency_manifest", path, name)]));
+    if (name === "go.mod") addTechnology(map, tech("go", "Go", "language", [provenance("dependency_manifest", path, name)]));
+    inspectProjectFile(path, content, map);
+    inspectStaticFile(path, content, map);
+  }
+
+  const detected = [...map.values()].sort((a, b) => a.id.localeCompare(b.id));
+  return {
+    discoveryId: stableId("disc", {
+      projectId: snapshot.projectId,
+      targetId: snapshot.targetId,
+      sourceRevision: snapshot.sourceRevision,
+      files: snapshot.files.map((file) => [file.path, Math.max(Buffer.byteLength(file.content ?? "", "utf8"), file.sizeBytes ?? 0), file.binary === true, file.kind ?? "file"]),
+      detected: detected.map((item) => item.id),
+      unsupported,
+    }),
+    projectId: snapshot.projectId,
+    targetId: snapshot.targetId,
+    sourceRevision: snapshot.sourceRevision,
+    detectorVersion: DETECTOR_VERSION,
+    readOnly: true,
+    detected,
+    unsupported,
+    unknown: detected.length === 0,
+    incomplete: false,
+    safety: {
+      packageScriptsExecuted: false,
+      networkUsed: false,
+      dependencyInstallAttempted: false,
+      arbitraryHostPathsAccepted: false,
+    },
+  };

+export const MAX_SECURITY_REQUEST_BODY_BYTES = 16 * 1024;

+export async function readBoundedSecurityJson(request: Request, maxBytes = MAX_SECURITY_REQUEST_BODY_BYTES): Promise<unknown> {
+  const declared = request.headers.get("content-length");
+  if (declared !== null) {
+    const value = Number(declared);
+    if (!Number.isSafeInteger(value) || value < 0 || value > maxBytes) throw new Error("security_request_body_too_large");
+  }
+
+  if (!request.body) throw new Error("security_request_invalid");
+
+  const reader = request.body.getReader();
+  const chunks: Uint8Array[] = [];
+  let total = 0;
+
+  try {
+    while (true) {
+      const { done, value } = await reader.read();
+      if (done) break;
+      if (!value) continue;
+      total += value.byteLength;
+      if (total > maxBytes) {
+        await reader.cancel("security_request_body_too_large");
+        throw new Error("security_request_body_too_large");
+      }
+      chunks.push(value);
+    }
+  } finally {
+    reader.releaseLock();
+  }
+
+  if (total === 0) throw new Error("security_request_invalid");
+
+  const bytes = new Uint8Array(total);
+  let offset = 0;
+  for (const chunk of chunks) {
+    bytes.set(chunk, offset);
+    offset += chunk.byteLength;
+  }
+
+  try {
+    return JSON.parse(new TextDecoder("utf-8", { fatal: true }).decode(bytes)) as unknown;
+  } catch {
+    throw new Error("security_request_invalid");
+  }
+}

+function objectBody(value: unknown): Record<string, unknown> {
+  if (!value || typeof value !== "object" || Array.isArray(value)) throw new Error("security_request_invalid");
+  return value as Record<string, unknown>;
+}

+function rejectUnknownKeys(body: Record<string, unknown>, allowed: readonly string[]): void {
+  const allowedSet = new Set(allowed);
+  if (Object.keys(body).some((key) => !allowedSet.has(key))) throw new Error("security_request_invalid");
+}

+export function parseSecurityDiscoveryRequest(value: unknown): SecurityDiscoveryRequest {
+  const body = objectBody(value);
+  rejectUnknownKeys(body, ["projectId", "targetId", "assessmentIntent"]);
+  if (typeof body.projectId !== "string" || !PROJECT_ID.test(body.projectId)) throw new Error("security_project_invalid");
+  if (body.targetId !== undefined && (typeof body.targetId !== "string" || !TARGET_ID.test(body.targetId))) throw new Error("security_target_invalid");
+  if (body.assessmentIntent !== undefined && !["source", "runtime", "pipeline"].includes(String(body.assessmentIntent))) {
+    throw new Error("security_request_invalid");
+  }
+  return {
+    projectId: body.projectId,
+    ...(typeof body.targetId === "string" ? { targetId: body.targetId } : {}),
+    ...(typeof body.assessmentIntent === "string" ? { assessmentIntent: body.assessmentIntent as SecurityDiscoveryRequest["assessmentIntent"] } : {}),
+  };
+}

+export function parseDiscoveryIdRequest(value: unknown): string {
+  const body = objectBody(value);
+  rejectUnknownKeys(body, ["discoveryId"]);
+  if (typeof body.discoveryId !== "string" || !DISCOVERY_ID.test(body.discoveryId)) throw new Error("security_discovery_id_invalid");
+  return body.discoveryId;

+  private async eligible(session: ProductSession): Promise<EligibleSecurityContext> {

+    }

+      organizationId: workspace.organizationId,
+      workspaceId: workspace.id,

+      entry.productUserId !== context.productUserId ||
+      entry.organizationId !== context.organizationId ||
+      entry.workspaceId !== context.workspaceId ||

+  private readonly entries = new Map<string, StoredSecurityDiscovery>();
+
+  constructor(private readonly maxEntries = 256) {}
+
+  put(entry: StoredSecurityDiscovery): void {
+    if (this.entries.size >= this.maxEntries && !this.entries.has(entry.result.discoveryId)) {
+      const first = this.entries.keys().next().value as string | undefined;
+      if (first) this.entries.delete(first);
+    }
+    this.entries.set(entry.result.discoveryId, entry);
+  }
+
+  get(discoveryId: string): StoredSecurityDiscovery | undefined {
+    return this.entries.get(discoveryId);
+  }
